    // ---------- Helpers ----------
    const $ = (id) => document.getElementById(id);
    function setStatus(kind, title, detail){
      const dot = $("statusDot");
      dot.classList.remove("good","bad");
      if(kind === "good") dot.classList.add("good");
      if(kind === "bad") dot.classList.add("bad");
      const text = `<b>${escapeHtml(title)}</b> ${escapeHtml(detail || "")}`;
      $("statusText").innerHTML = text;
    }
    function escapeHtml(str){
      return (str||"").replaceAll("&","&amp;").replaceAll("<","&lt;").replaceAll(">","&gt;");
    }
    function mergedOnScreenText(){
      return SCENES.map(s => s.text).join("\n\n");
    }
    function makeMergedPrompt(platform){
      const aspect = $("aspect").value;
      const duration = $("duration").value;
      const music = $("music").value;
      const style = $("style").value.trim();

      const lockRule =
`CRITICAL TEXT LOCK:
All on-screen text below MUST appear exactly as written.
Do NOT rewrite, paraphrase, summarize, correct, or add narration.
Do NOT reorder lines.`;

      const header =
`Create a ${duration} cinematic B2B product explainer video for:
"HubSpot Revenue Rescue — No-Nonsense Growth Agent"
Aspect ratio: ${aspect}
Music: ${music}`;

      const body = `ON-SCREEN TEXT (LOCKED):\n\n${mergedOnScreenText()}`;

      // Platform-specific slight tuning (still one-file, deterministic)
      if(platform === "nano"){
        return [
          "NANO BANANAS — LOCKED VIDEO PROMPT",
          header,
          lockRule,
          "VISUAL STYLE:",
          style,
          "",
          "SCENE STRUCTURE:",
          SCENES.map(s =>
            `Scene ${s.id} — ${s.title} (${s.time})\nOn-screen text:\n${s.text}\nVisual:\n${s.visual}`
          ).join("\n\n---\n\n")
        ].join("\n\n");
      }

      if(platform === "pippit"){
        return [
          header,
          lockRule,
          "STYLE:",
          style,
          "",
          body,
          "",
          "Optional audio: subtle, low-tempo, modern background music only."
        ].join("\n\n");
      }

      if(platform === "canva"){
        return [
          header,
          lockRule,
          "STYLE:",
          style,
          "",
          "Use one scene per text block. Do not modify text.",
          "",
          body
        ].join("\n\n");
      }

      // gamma
      return [
        header,
        lockRule,
        "STYLE:",
        style,
        "",
        body
      ].join("\n\n");
    }

    function makeScenesBlocks(){
      return SCENES.map(s => {
        return [
          `SCENE ${s.id} — ${s.title} (${s.time})`,
          "ON-SCREEN TEXT (LOCKED):",
          s.text,
          "",
          "VISUAL DESCRIPTION:",
          s.visual
        ].join("\n");
      }).join("\n\n====================\n\n");
    }

    async function copyToClipboard(text){
      try{
        await navigator.clipboard.writeText(text);
        return true;
      }catch(e){
        // fallback
        const ta = document.createElement("textarea");
        ta.value = text;
        document.body.appendChild(ta);
        ta.select();
        const ok = document.execCommand("copy");
        document.body.removeChild(ta);
        return ok;
      }
    }

    function download(filename, content, mime="application/json"){
      const blob = new Blob([content], {type:mime});
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
    }
